Practical No-2 [configuration]
git --version
git config --global user.name "Your Name" // config username 
git config --global user.email "your@email.com" // config email id 
git config --global core.editor "code --wait" // setup vs studio for coding 
git config --list // check whether set up is done properly 
notepad ~/.gitconfig // open config details on notepad 

practical no-3 [ssh keys]
ssh-keygen -t ed25519 -C "your@email.com"  // create ssh key 
eval "$(ssh-agent -s)" // starts ssh 
ssh-add ~/.ssh/id_ed25519 // adds ssh key 
ssh -T git@github.com // test GitHub connection 
cat ~/.ssh/id_ed25519.pub // display ssh public key
ssh -T git@github.com // test connection again
